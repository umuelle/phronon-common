# Test-only dependencies. Not runtime dependencies of the package.
#
# phronon_common is deliberately framework-agnostic — pyproject declares only
# itsdangerous, and the tools bring their own FastAPI/Starlette. But csrf.py,
# rate_limit.py and security_headers.py ship Starlette middleware, and the only
# honest way to test middleware is to mount it on a real ASGI app and drive it
# with a real client. Pinned to the versions the fleet deploys, so the middleware
# is tested against the Starlette the tools actually run.
fastapi==0.139.2
starlette==1.3.1

# httpx2, NOT httpx — and this one is load-bearing.
#
# starlette 1.3.1's TestClient warns "Using `httpx` with `starlette.testclient`
# is deprecated; install `httpx2`". On the httpx-1.x path the client does not
# deliver a form body to `await request.form()` inside a BaseHTTPMiddleware, so
# a POST carrying a valid csrf_token FORM FIELD is seen by the middleware as
# having no token at all, and the test fails with a 403. That is a test-harness
# artifact, not a product bug: the same request against a real uvicorn server on
# starlette 1.3.1 returns 200 (verified 2026-07-28 — form token 200, header
# token 200, no token 403).
#
# It is exactly the kind of failure worth pinning against, because the naive
# reading is "form CSRF is broken, weaken the test to use the header instead",
# which would delete the coverage that matters: every backoffice form in the
# fleet submits its token as a form field.
httpx2==2.9.1
