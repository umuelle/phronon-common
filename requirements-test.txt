# Test-only dependencies. Not runtime dependencies of the package.
#
# phronon_common is deliberately framework-agnostic — pyproject declares only
# itsdangerous, and the tools bring their own FastAPI/Starlette. But csrf.py,
# rate_limit.py and security_headers.py ship Starlette middleware, and the only
# honest way to test middleware is to mount it on a real ASGI app and drive it
# with a real client. Pinned to the versions the fleet deploys, so the middleware
# is tested against the Starlette the tools actually run.
fastapi==0.139.2
starlette==1.3.1
# 2.9.1 → 2.12.0 on 10 September 2026: CI's pip-audit turned red on six
# advisories against 2.9.1 (PYSEC-2026-3845…3849) and its httpcore2
# (PYSEC-2026-3844); 2.12.0 is the first version clear of all of them.
# httpcore2 is pinned explicitly for the same reason — it is httpx2's own
# dependency, but the audit reads the resolved graph, so it has to be named.
httpx2==2.13.1
httpcore2==2.12.0

# python-multipart is NOT optional here, and the failure it causes is misleading.
#
# Without it, `await request.form()` inside the CSRF middleware yields nothing,
# so a POST carrying a perfectly valid csrf_token FORM FIELD looks to the
# middleware like a request with no token at all and comes back 403 — while the
# X-CSRF-Token header path keeps working. It reads exactly like "form CSRF is
# broken", and the tempting fix is to weaken the tests to send a header instead.
# That would delete the coverage that matters: every backoffice form in the
# fleet submits its token as a form field.
#
# The tools never hit this because python-multipart is in their requirements.txt
# (FastAPI needs it for form handling). This package has no requirements.txt, so
# it has to be named here. Verified 2026-07-28 on python 3.10 / starlette 1.3.1:
# without it 2 tests fail, with it all 18 pass, on both httpx and httpx2.
python-multipart==0.0.32

# passwords.py and twofactor.py import bcrypt, and pyproject declares only
# itsdangerous, so nothing installs it here. CI has been RED on this since at
# least 64ec06c: test_password_length_bound.py fails at COLLECTION, which takes
# the whole run down — so every other test in this package stopped gating too,
# silently, while the tools kept passing because their own requirements.txt
# names bcrypt. Same version the fleet deploys (all nine pin 5.0.0), because
# the 72-byte bound this package asserts is bcrypt's, not ours.
bcrypt==5.0.0

# legal.py renders the fleet legal pages through its own Jinja
# environment. The tools all ship jinja2 via FastAPI templating; this
# package has no requirements.txt, so the test environment names it.
jinja2==3.1.6

# passkeys.py and testing/soft_authenticator.py (FL-065, 2 October 2026) use
# py_webauthn, which brings cryptography and cbor2. It is in the `web` extra,
# but CI installs this package with `pip install -e .` (core only), so it is
# named here; the version the nine tools deploy. tests/test_passkeys.py does
# NOT importorskip it: a missing library must fail the run, not skip quietly.
webauthn==3.0.1

# The DB-backed tests (passkey one-use rule, once() under real row locks) need
# a real MySQL. CI provides one since 3 October 2026 and sets
# PHRONON_REQUIRE_DB=1, so a missing connector FAILS there instead of skipping.
mysql-connector-python==9.7.0
